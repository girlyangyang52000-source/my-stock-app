async function autoFetchMarketPrices() { const { holdingBuyIds } = calculateAutoStatus(records); const holdingItems = records.filter(r => r.type === 'BUY' && holdingBuyIds.has(String(r.id))); if (holdingItems.length === 0) { alert("目前沒有庫存股票可抓取！"); return; } // 收集所有庫存股票對應的代號 const targetCodes = []; holdingItems.forEach(r => { const cleanName = r.name.trim().replace(/^[0-9A-Za-z]+/, '').trim(); const code = STOCK_CODE_MAP[r.name.trim()] || STOCK_CODE_MAP[cleanName]; if (code && !targetCodes.includes(code)) { targetCodes.push(code); } }); if (targetCodes.length === 0) { alert("找不到庫存股票的對應代號，請確認股票名稱是否在對照表中。"); return; } const btnIcon = document.getElementById('fetchIcon'), btnText = document.getElementById('fetchText'); btnIcon.classList.add('animate-spin'); btnText.innerText = "正在向證交所 API 請求..."; try { const res = await fetch(`/stock?codes=${targetCodes.join(',')}`); const result = await res.json(); if (result.success && Array.isArray(result.data)) { const priceMap = {}; result.data.forEach(item => { if (item.Code && item.ClosingPrice) { priceMap[String(item.Code).trim()] = parseFloat(String(item.ClosingPrice).replace(/,/g, '')); } }); let updatedCount = 0; holdingItems.forEach(r => { const cleanName = r.name.trim().replace(/^[0-9A-Za-z]+/, '').trim(); const code = STOCK_CODE_MAP[r.name.trim()] || STOCK_CODE_MAP[cleanName]; const livePrice = code ? priceMap[code] : null; if (livePrice && !isNaN(livePrice) && livePrice > 0) { livePriceCache[r.id] = String(livePrice); const inp = document.getElementById(`live-input-${r.id}`); if (inp) inp.value = livePrice; const discount = r.discount !== undefined ? r.discount : 0.3; onLivePriceChange(r.id, Math.abs(parseFloat(r.totalAmount) || 0), getRecordShares(r), discount, String(livePrice)); updatedCount++; } }); btnText.innerText = updatedCount > 0 ? `成功抓取 ${updatedCount} 檔時價` : "未在 API 找到對應代號"; } else { btnText.innerText = "API 回應格式異常"; } } catch (err) { btnText.innerText = "無法連線至後端 API"; } finally { btnIcon.classList.remove('animate-spin'); setTimeout(() => { btnText.innerText = "從證交所 API 抓取即時時價"; }, 3000); } }
