async function autoFetchMarketPrices() { const { holdingBuyIds } = calculateAutoStatus(records); const holdingItems = records.filter(r => r.type === 'BUY' && holdingBuyIds.has(String(r.id))); if (holdingItems.length === 0) { alert("目前沒有庫存股票可抓取！"); return; } const btnIcon = document.getElementById('fetchIcon'), btnText = document.getElementById('fetchText'); btnIcon.classList.add('animate-spin'); btnText.innerText = "正在連線取得即時市價..."; let updatedCount = 0; for (const r of holdingItems) { const cleanName = r.name.trim().replace(/^[0-9A-Za-z]+/, '').trim(); const code = STOCK_CODE_MAP[r.name.trim()] || STOCK_CODE_MAP[cleanName]; if (!code) continue; try { // 使用完全公開且支援 CORS 的證交所盤後/即時公開資料接口 let res = await fetch(`https://www.twse.com.tw/exchangeReport/STOCK_DAY_ALL?response=json`); let json = await res.json(); let foundPrice = null; if (json && Array.isArray(json.data)) { // json.data 格式: [代號, 名稱, 成交股數, 成交金額, 開盤, 最高, 最低, 收盤, ...] const stockRow = json.data.find(row => row[0] === code); if (stockRow && stockRow[7]) { foundPrice = parseFloat(stockRow[7].replace(/,/g, '')); } } // 如果上市找不到，改查櫃買中心上櫃公開接口 if (!foundPrice) { let tpexRes = await fetch(`https://www.tpex.org.tw/openapi/v1/tpex_mainboard_quotes`); let tpexJson = await tpexRes.json(); if (Array.isArray(tpexJson)) { const tpexRow = tpexJson.find(row => (row.SecuritiesCompanyCode === code || row.Code === code)); if (tpexRow && (tpexRow.Close || tpexRow.ClosingPrice)) { foundPrice = parseFloat((tpexRow.Close || tpexRow.ClosingPrice).toString().replace(/,/g, '')); } } } if (foundPrice && !isNaN(foundPrice) && foundPrice > 0) { livePriceCache[r.id] = String(foundPrice); const inp = document.getElementById(`live-input-${r.id}`); if (inp) inp.value = foundPrice; const discount = r.discount !== undefined ? r.discount : 0.3; onLivePriceChange(r.id, Math.abs(parseFloat(r.totalAmount) || 0), getRecordShares(r), discount, String(foundPrice)); updatedCount++; } } catch (e) { console.error("Fetch error:", e); } } btnIcon.classList.remove('animate-spin'); btnText.innerText = updatedCount > 0 ? `成功抓取 ${updatedCount} 檔時價` : "已完成抓取"; setTimeout(() => { btnText.innerText = "從財經 API 動態抓取即時時價"; }, 3000); }
